def rename_matching(directory, keyword, target):
    files = os.listdir(directory)
    matches = [f for f in files if keyword in f]

    if target in files:
        print("TARGET ALREADY EXISTS:", os.path.join(directory, target))
        return

    if not matches:
        raise Exception(f"NOT FOUND: {keyword} in {directory}")

    if len(matches) > 1:
        raise Exception(f"MULTIPLE MATCHES: {matches}")

    source = matches[0]
    print(f"RENAMING: {source} -> {target}")
    os.rename(
        os.path.join(directory, source),
        os.path.join(directory, target)
    )
